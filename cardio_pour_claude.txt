const crypto = require("crypto");
const mongoose = require("mongoose");

const sha = (s) => crypto.createHash("sha256").update(String(s)).digest("hex");
const uid = (req) => String(req.user._id || req.user.id);
const ALPHA = "ABCDEFGHJKLMNPQRSTUVWXYZ23456789"; // sans I, O, 0, 1
const introuvable = (res) => res.status(404).json({ message: "Introuvable." });

// limite les essais de code d'appairage : 10 par 10 minutes et par adresse IP
const essais = new Map();
function autorise(ip) {
  const now = Date.now();
  const e = essais.get(ip);
  if (!e || e.reset < now) {
    essais.set(ip, { n: 1, reset: now + 10 * 60 * 1000 });
    return true;
  }
  e.n += 1;
  return e.n <= 10;
}

exports.deviceStatus = async (req, res, next) => {
  try {
    const pid = uid(req);
    const dev = await CardioDevice.findOne({ patient: pid });
    res.json({
      paired: !!(dev && dev.keyHash),
      online: isDeviceOnline(pid),
      lastSeen: dev ? dev.lastSeen : null,
      pairingActive: !!(dev && dev.pairingExpires && dev.pairingExpires > new Date()),
    });
  } catch (e) { next(e); }
};

exports.createPairing = async (req, res, next) => {
  try {
    const raw = Array.from(crypto.randomBytes(6), (b) => ALPHA[b % 32]).join("");
    const expiresAt = new Date(Date.now() + 10 * 60 * 1000);
    await CardioDevice.findOneAndUpdate(
      { patient: uid(req) },
      { $set: { pairingHash: sha(raw), pairingExpires: expiresAt } },
      { upsert: true, setDefaultsOnInsert: true }
    );
    res.json({ code: `${raw.slice(0, 3)}-${raw.slice(3)}`, expiresAt });
  } catch (e) { next(e); }
};

// AppelÃ© par le Raspberry Pi (sans JWT) : Ã©change le code contre une clÃ© durable
exports.pair = async (req, res, next) => {
  try {
    if (!autorise(req.ip)) {
      return res.status(429).json({ message: "Trop d'essais. RÃ©essayez dans quelques minutes." });
    }
    const raw = String((req.body && req.body.code) || "").toUpperCase().replace(/[^A-Z0-9]/g, "");
    if (raw.length !== 6) return res.status(400).json({ message: "Code invalide ou expirÃ©." });

    const cle = crypto.randomBytes(32).toString("hex");
    const dev = await CardioDevice.findOneAndUpdate(
      { pairingHash: sha(raw), pairingExpires: { $gt: new Date() } },
      { $set: { keyHash: sha(cle) }, $unset: { pairingHash: "", pairingExpires: "" } },
      { new: true }
    );
    if (!dev) return res.status(400).json({ message: "Code invalide ou expirÃ©." });

    disconnectDevice(dev.patient);
    res.json({ deviceKey: cle });
  } catch (e) { next(e); }
};

exports.removeDevice = async (req, res, next) => {
  try {
    const pid = uid(req);
    await CardioDevice.updateOne(
      { patient: pid },
      { $unset: { keyHash: "", pairingHash: "", pairingExpires: "", lastSeen: "" } }
    );
    const s = await CardioSession.findOneAndUpdate(
      { patient: pid, status: "active" },
      { $set: { status: "ended", endedAt: new Date() } },
      { new: true }
    );
    if (s) notifySession(pid, null);
    disconnectDevice(pid);
    res.json({ ok: true });
  } catch (e) { next(e); }
};

exports.startSession = async (req, res, next) => {
  try {
    const pid = uid(req);
    if (!isDeviceOnline(pid)) {
      return res.status(409).json({ message: "Le dispositif n'est pas connectÃ©." });
    }
    let s = await CardioSession.findOne({ patient: pid, status: "active" });
    if (!s) {
      try {
        s = await CardioSession.create({ patient: pid, startedAt: new Date(), status: "active" });
      } catch (e) {
        if (e.code !== 11000) throw e;
        s = await CardioSession.findOne({ patient: pid, status: "active" });
      }
    }
    notifySession(pid, s);
    res.status(201).json({ session: s });
  } catch (e) { next(e); }
};

exports.listSessions = async (req, res, next) => {
  try {
    const sessions = await CardioSession.find({ patient: uid(req) }).sort({ startedAt: -1 }).limit(50).lean();
    const agg = await CardioMeasure.aggregate([
      { $match: { session: { $in: sessions.map((s) => s._id) } } },
      { $group: { _id: "$session", count: { $sum: 1 }, bpm: { $avg: "$bpm" }, stress: { $avg: "$stress" }, rmssd: { $avg: "$rmssd" } } },
    ]);
    const par = new Map(agg.map((a) => [String(a._id), a]));
    res.json(
      sessions.map((s) => {
        const a = par.get(String(s._id));
        return {
          ...s,
          count: a ? a.count : 0,
          bpmAvg: a ? Math.round(a.bpm) : null,
          stressAvg: a ? Math.round(a.stress) : null,
          rmssdAvg: a ? Math.round(a.rmssd * 10) / 10 : null,
        };
      })
    );
  } catch (e) { next(e); }
};

const sha = (s) => crypto.createHash("sha256").update(String(s)).digest("hex");
const fin = (v) => typeof v === "number" && Number.isFinite(v);
const tableau = (a, max) => Array.isArray(a) && a.length <= max && a.every(fin);
const dans = (v, lo, hi) => fin(v) && v >= lo && v <= hi;
const texte = (s) => (typeof s === "string" ? s.slice(0, 200) : "");

const live = new Map(); // patientId -> { online, state, metrics }
let nsDevice = null;
let nsPatient = null;

const versPatient = (pid, evt, data) => nsPatient && nsPatient.to(`patient:${pid}`).emit(evt, data);

const isDeviceOnline = (pid) => !!(live.get(String(pid)) && live.get(String(pid)).online);

function notifySession(pid, session) {
  pid = String(pid);
  if (nsDevice) {
    nsDevice.to(`device:${pid}`).emit("session:sync", { sessionId: session ? String(session._id) : null });
  }
  versPatient(pid, "session:update", { session: session || null });
}

function disconnectDevice(pid) {
  if (nsDevice) nsDevice.in(`device:${String(pid)}`).disconnectSockets(true);

const mpuPropre = (m) =>

function initCardio(io) {
  nsDevice = io.of("/device");
  nsPatient = io.of("/cardio");

  /* ---------- Raspberry Pi ---------- */
  nsDevice.use(async (socket, next) => {
    try {
      const cle = socket.handshake.auth && socket.handshake.auth.deviceKey;
      if (!cle) return next(new Error("unauthorized"));
      const dev = await CardioDevice.findOne({ keyHash: sha(cle) }).select("patient");
      if (!dev) return next(new Error("unauthorized"));
      socket.data.patientId = String(dev.patient);
      next();
    } catch {
      next(new Error("server"));
    }
  });

  nsDevice.on("connection", async (socket) => {
    const pid = socket.data.patientId;
    const room = `device:${pid}`;
    try {
      // un seul Pi par patient : le plus rÃ©cent remplace l'ancien
      const anciens = await nsDevice.in(room).fetchSockets();
      anciens.forEach((s) => s.disconnect(true));
      socket.join(room);
      live.set(pid, { online: true, state: null, metrics: null });
      await CardioDevice.updateOne({ patient: pid }, { $set: { lastSeen: new Date() } });
      versPatient(pid, "device:status", { online: true });

      const active = await CardioSession.findOne({ patient: pid, status: "active" }).select("_id");
      socket.emit("session:sync", { sessionId: active ? String(active._id) : null });
    } catch (e) {
      console.error("[cardio] connexion Pi :", e.message);
    }

    socket.on("signal", (d) => {
      if (!d || !tableau(d.y, 4000) || !tableau(d.pics || [], 300) || !fin(d.fs)) return;
      versPatient(pid, "signal", { y: d.y, pics: d.pics || [], fs: d.fs });
    });

    socket.on("state", (d) => {
      if (!d || typeof d !== "object") return;
      const state = {
        signal_recu: !!d.signal_recu,
        electrodes_ok: !!d.electrodes_ok,
        mpu: !!d.mpu,
        simulation: !!d.simulation,
      };
      const l = live.get(pid);
      if (l) l.state = state;
      versPatient(pid, "device:state", state);
    });

    socket.on("metrics", (d) => {
      if (!d || typeof d !== "object") return;
      const m = metriquesPropres(d);
      const l = live.get(pid);
      if (l) l.metrics = m;
      versPatient(pid, "metrics", m);
    });

    socket.on("measure", async (d) => {
      try {
        if (!d || !mongoose.isValidObjectId(d.session_id)) return;
        const s = await CardioSession.findOne({ _id: d.session_id, patient: pid, status: "active" }).select("_id");
        if (!s) return;
        if (!dans(d.bpm, 20, 250) || !dans(d.stress, 0, 100) || !dans(d.rmssd, 0, 1000)) return;
        const doc = await CardioMeasure.create({
          session: s._id,
          patient: pid,
          at: new Date(),
          bpm: d.bpm,
          rmssd: d.rmssd,
          sdnn: fin(d.sdnn) ? d.sdnn : null,
          pnn50: fin(d.pnn50) ? d.pnn50 : null,
          stress: d.stress,
          level: ["Faible", "ModÃ©rÃ©", "Ã‰levÃ©"].includes(d.niveau) ? d.niveau : null,
          beats: fin(d.nb_battements) ? d.nb_battements : null,
          movement: fin(d.mouvement) ? d.mouvement : null,
          ecg: tableau(d.ecg, 2500) ? d.ecg : [],
        });
        const { ecg, ...legere } = doc.toObject();
        versPatient(pid, "measure:saved", legere);
      } catch (e) {
        console.error("[cardio] mesure :", e.message);
      }
    });

    socket.on("disconnect", async () => {
      const restants = await nsDevice.in(room).fetchSockets();
      if (restants.length === 0) {
        live.delete(pid);
        versPatient(pid, "device:status", { online: false });
      }
    });
  });

  /* ---------- Navigateur du patient ---------- */
  nsPatient.use((socket, next) => {
    try {
      const p = jwt.verify(socket.handshake.auth && socket.handshake.auth.token, process.env.JWT_SECRET);
      const id = p.id || p._id || p.userId || p.sub;
      if (!id) throw new Error("no id");
      socket.data.patientId = String(id);
      next();
    } catch {
      next(new Error("unauthorized"));
    }
  });

  nsPatient.on("connection", (socket) => {
    const pid = socket.data.patientId;
    socket.join(`patient:${pid}`);
    const l = live.get(pid);
    socket.emit("device:status", { online: !!(l && l.online) });
    if (l && l.state) socket.emit("device:state", l.state);
    if (l && l.metrics) socket.emit("metrics", l.metrics);
  });
}

const jwt = require("jsonwebtoken");

const ApiError = require("../utils/ApiError");

// VÃ©rifie le token JWT et charge l'utilisateur dans req.user
const protect = async (req, res, next) => {
  const header = req.headers.authorization;
  if (!header || !header.startsWith("Bearer ")) {
    throw new ApiError(401, "Authentification requise");
  }

  let decoded;
  try {
    decoded = jwt.verify(header.split(" ")[1], process.env.JWT_SECRET);
  } catch (error) {
    throw new ApiError(401, "Token invalide ou expirÃ©");
  }

  const user = await User.findById(decoded.id);
  if (!user || !user.estActif) {
    throw new ApiError(401, "Compte introuvable ou suspendu");
  }

  req.user = user;
  next();
};

module.exports = { protect };

import { useEffect, useState } from "react";
import { io } from "socket.io-client";
import { getToken } from "../services/cardioApi";

// Adresse du backend (sans /api)
const SOCKET_URL = import.meta.env.VITE_SOCKET_URL || "http://localhost:5000";

export default function useCardioLive() {
  const [live, setLive] = useState({
    socketUp: false,
    deviceOnline: false,
    state: null,
    metrics: null,
    signal: null,
    signalAt: 0,
    stateAt: 0,
  });
  const [lastMeasure, setLastMeasure] = useState(null);
  const [sessionEvt, setSessionEvt] = useState(null);

  useEffect(() => {
    const socket = io(`${SOCKET_URL}/cardio`, { auth: { token: getToken() } });
    const set = (patch) => setLive((s) => ({ ...s, ...patch }));

    socket.on("connect", () => set({ socketUp: true }));
    socket.on("disconnect", () => set({ socketUp: false, deviceOnline: false }));
    socket.on("connect_error", () => set({ socketUp: false }));

    socket.on("device:status", (d) =>
      set(
        d.online
          ? { deviceOnline: true }
          : { deviceOnline: false, signal: null, metrics: null, state: null }
      )
    );
    socket.on("device:state", (d) => set({ state: d, stateAt: Date.now() }));
    socket.on("signal", (d) => set({ signal: d, signalAt: Date.now() }));
    socket.on("metrics", (d) => set({ metrics: d }));
    socket.on("measure:saved", (m) => setLastMeasure(m));
    socket.on("session:update", (d) => setSessionEvt({ ...d, at: Date.now() }));

    return () => socket.disconnect();
  }, []);

  return { ...live, lastMeasure, sessionEvt };
}

function useCanvas(draw, deps) {
  const ref = useRef(null);
  useEffect(() => {
    const c = ref.current;
    if (!c) return;
    const dpr = window.devicePixelRatio || 1;
    const w = c.clientWidth;
    const h = c.clientHeight;
    c.width = Math.round(w * dpr);
    c.height = Math.round(h * dpr);
    const ctx = c.getContext("2d");
    ctx.setTransform(dpr, 0, 0, dpr, 0, 0);
    draw(ctx, w, h);
    // eslint-disable-next-line react-hooks/exhaustive-deps
  }, deps);
  return ref;
}

function fond(ctx, w, h, pas = 24) {
  ctx.fillStyle = "#fff";
  ctx.fillRect(0, 0, w, h);
  ctx.strokeStyle = "#f3d6d3";
  ctx.lineWidth = 0.6;
  for (let x = 0; x <= w; x += pas) {
    ctx.beginPath();
    ctx.moveTo(x, 0);
    ctx.lineTo(x, h);
    ctx.stroke();
  }
  for (let y = 0; y <= h; y += pas) {
    ctx.beginPath();
    ctx.moveTo(0, y);
    ctx.lineTo(w, y);
    ctx.stroke();
  }
}

function texte(ctx, t) {
  ctx.fillStyle = "#6b7a76";
  ctx.font = "15px system-ui";
  ctx.fillText(t, 16, 28);
}

function dessinEcg(ctx, w, h, y, pics) {
  fond(ctx, w, h);
  if (!y || y.length < 2) return texte(ctx, "En attente du signalâ€¦");
  const n = y.length;
  const moy = y.reduce((a, b) => a + b, 0) / n;
  const sd = Math.sqrt(y.reduce((a, b) => a + (b - moy) ** 2, 0) / n);
  let lo = Math.max(Math.min(...y), moy - 5 * sd);
  let hi = Math.min(Math.max(...y), moy + 5 * sd);
  if (hi - lo < 10) {
    const c = (hi + lo) / 2;
    lo = c - 5;
    hi = c + 5;
  }
  const py = (v) => h * 0.92 - ((Math.min(Math.max(v, lo), hi) - lo) / (hi - lo)) * h * 0.84;
  const px = (i) => (i / (n - 1)) * w;

  ctx.strokeStyle = "#1b2b27";
  ctx.lineWidth = 1.6;
  ctx.lineJoin = "round";
  ctx.beginPath();
  y.forEach((v, i) => (i ? ctx.lineTo(px(i), py(v)) : ctx.moveTo(px(i), py(v))));
  ctx.stroke();

  ctx.fillStyle = "#e5484d";
  (pics || []).forEach((i) => {
    if (i < n) {
      ctx.beginPath();
      ctx.arc(px(i), py(y[i]), 4, 0, 6.2832);
      ctx.fill();
    }
  });
}

function dessinCourbes(ctx, w, h, series, vide) {
  fond(ctx, w, h, 40);
  const n = series[0] ? series[0].vals.length : 0;
  if (n < 2) return texte(ctx, vide);
  series.forEach(({ vals, lo, hi, couleur }) => {
    ctx.strokeStyle = couleur;
    ctx.lineWidth = 2;
    ctx.beginPath();
    vals.forEach((v, i) => {
      const x = 12 + (i / (n - 1)) * (w - 24);
      const vv = Math.min(Math.max(v ?? lo, lo), hi);
      const yy = h - 12 - ((vv - lo) / (hi - lo)) * (h - 24);
      if (i) ctx.lineTo(x, yy);
      else ctx.moveTo(x, yy);
    });
    ctx.stroke();
  });
}

const Courbes = ({ series, vide }) => {
  const ref = useCanvas((c, w, h) => dessinCourbes(c, w, h, series, vide), [series]);
  return <canvas ref={ref} className="block h-48 w-full rounded-xl border border-rose-100" />;
};
